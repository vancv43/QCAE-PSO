\begin{algorithm}[H]
\caption{PSO for QCAE Hyperparameter Optimization}
\label{alg:pso}
\begin{algorithmic}[1]
\Require Training split $D_{train}$, validation split $D_{val}$, population size $N$, iterations $T$, bounds $[0,1]^4$, fitness function $\mathcal{F}$, inertia weight $w$, cognitive coefficient $c_1$, social coefficient $c_2$
\Ensure Best hyperparameter vector $x^{*}$
\State Initialize positions $\{x_i\}_{i=1}^{N}$ uniformly over $[0,1]^4$; set velocities $v_i \leftarrow 0$
\State Train and evaluate each $x_i$ via $\mathcal{F}$; set personal bests $p_i \leftarrow x_i$
\State Set global best $x^{*} \leftarrow p_i$ for the $i$ minimizing $\mathcal{F}(p_i)$
\For{$t = 1$ to $T$}
    \For{each particle $i = 1, \dots, N$}
        \State Draw random $r_1, r_2 \sim U(0,1)^4$
        \State Update velocity: $v_i \leftarrow w\,v_i + c_1 r_1 \odot (p_i - x_i) + c_2 r_2 \odot (x^{*} - x_i)$
        \State Update position: $x_i \leftarrow \mathrm{clip}(x_i + v_i, [0,1]^4)$
    \EndFor
    \For{each particle $i = 1, \dots, N$}
        \State Decode $x_i$, train the network, and evaluate $\mathcal{F}(x_i)$
        \If{$\mathcal{F}(x_i) < \mathcal{F}(p_i)$}
            \State Update personal best $p_i \leftarrow x_i$
        \EndIf
    \EndFor
    \State Update global best $x^{*} \leftarrow p_i$ for the $i$ minimizing $\mathcal{F}(p_i)$, if improved
\EndFor
\State \Return $x^{*}$
\end{algorithmic}
\end{algorithm}
